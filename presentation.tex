      \begin{tcolorbox}[darkcard, height=3.45cm, valign=top, top=4pt, bottom=4pt]
        {\fontsize{8.5pt}{9.5pt}\selectfont\color{cyanGlow}\textbf{Bisection Root-Finding Principle}}\\[2pt]
        {\fontsize{7.5pt}{8.5pt}\selectfont
        $\bullet$ \textbf{Objective}: Root of signed error $\Delta D(T) \to 0$\\[1.5pt]
        \quad $\Delta D(T) = \frac{1}{4}\sum_{k=1}^4 (D_{\text{pred}, k}(T) - D_{\text{gt}, k})$\\[1.5pt]
        $\bullet$ \textbf{Monotonicity}: Energy $T \uparrow \implies$ duration $D \downarrow$\\[1.5pt]
        $\bullet$ \textbf{Direction}: $\Delta D > 0 \implies T_{\min} = T$; else $T_{\max} = T$\\[1.5pt]
        $\bullet$ \textbf{Precision}: Interval halving $2^{-40} \approx 4.5 \times 10^{-14}$}
      \end{tcolorbox}
    \end{column}

    \begin{column}{0.48\textwidth}

        {\fontsize{7.5pt}{8.5pt}\selectfont
        $\bullet$ \textbf{Global threshold}: $\mathbf{T_{\text{opt}} = 0.003478 \approx 0.00348}$\\[1.5pt]
        $\bullet$ \textbf{Why signed error}? Boundary MAE is non-monotonic; bisection requires zero-crossing\\[1.5pt]
        $\bullet$ \textbf{Guaranteed root}: Monotonic search contract\\[1.5pt]
        $\bullet$ \textbf{Result}: Universal static threshold for all test audio}
      \end{tcolorbox}

  \frametitle{Rebuttal for Critical Examination Questions}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[darkcard, height=1.90cm, valign=top]
        {\fontsize{8pt}{9pt}\selectfont\color{cyanGlow}\textbf{Q1: Why not use F0 for boundary cuts?}}\\[2pt]
        {\fontsize{7.5pt}{8.5pt}\selectfont
        Unvoiced consonants (/s/, /t/, /f/) have no pitch F0. Using F0 would truncate initial and trailing phonemes!}
      \end{tcolorbox}
      \vskip0.15cm
      \begin{tcolorbox}[darkcard, height=1.90cm, valign=top]
        {\fontsize{8pt}{9pt}\selectfont\color{cyanGlow}\textbf{Q2: Why did phone\_M2 achieve 0.0 ms MAE?}}\\[2pt]
        {\fontsize{7.5pt}{8.5pt}\selectfont
        Labels [0.53s, 2.52s] coincide with frames 53 \& 252 (10 ms hop grid). It is exact grid snapping, not overfitting!}
      \end{tcolorbox}
    \end{column}

    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[darkcard, height=1.90cm, valign=top]
        {\fontsize{8pt}{9pt}\selectfont\color{cyanGlow}\textbf{Q3: Impact of acoustic noise floor (SNR)?}}\\[2pt]
        {\fontsize{7.5pt}{8.5pt}\selectfont
        Studio (high SNR): Error only 5--10 ms. Phone (low SNR): Trailing breath introduces 20--30 ms boundary lag.}
      \end{tcolorbox}
      \vskip0.15cm
      \begin{tcolorbox}[darkcard, height=1.90cm, valign=top]
        {\fontsize{8pt}{9pt}\selectfont\color{cyanGlow}\textbf{Q4: Why did 1D beat 4D Gaussian in TT3?}}\\[2pt]
        {\fontsize{7.5pt}{8.5pt}\selectfont
        Train set has only 4 files. 4D covariance matrix overfits; 1D energy model is robust and generalizes better!}
      \end{tcolorbox}
    \end{column}
  \end{columns}
